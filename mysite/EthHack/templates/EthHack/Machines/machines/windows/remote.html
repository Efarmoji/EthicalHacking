And wrote code

from Crypto.Cipher import AES

key = b"\x06\x02\x00\x00\x00\xa4\x00\x00\x52\x53\x41\x31\x00\x04\x00\x00"
iv = b"\x01\x00\x01\x00\x67\x24\x4F\x43\x6E\x67\x62\xF2\x5E\xA8\xD7\x04"
 
with open('/home/kali/tempo/AES.encrypted',mode='r',encoding='utf-8',errors='ignore') as f:
    list = [line.rstrip('\n') for line in f]
digits = []

for i in list:
    digit = int(i)
    digits.append(digit)

print(digits)
encryption = bytes(digits)
aes = AES.new(key, AES.MODE_CBC, IV=iv)
password = aes.decrypt(encryption).decode("utf-16").rstrip("\x00")

print(f"[:l] CRACKED PASSWORD! : {password}")
